Custom Rule

10+ years ago (custom)
const config = {
  rules: [{
    condition: function(dateInterval) {
      return dateInterval.day >= 365 * 10;
    },
    text: {
      fr: "%dd jour{%dd||s} (année %Y)",
      en: "%dd day{%dd||s} ago (year %Y)"
    }
  }]
};

DateJs. DomTimeago.start(config);
